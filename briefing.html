function doGet(e) {
  var cb = (e.parameter.callback || 'cb').replace(/[^a-zA-Z0-9_]/g, '');
  var out;
  try {
    var target = e.parameter.url;
    if (!target || !/^https:\/\//.test(target)) throw new Error('bad url');
    var resp = UrlFetchApp.fetch(target, { muteHttpExceptions: true, followRedirects: true });
    out = cb + '(' + JSON.stringify({ ok: true, status: resp.getResponseCode(), body: resp.getContentText() }) + ');';
  } catch (err) {
    out = cb + '(' + JSON.stringify({ ok: false, error: String(err) }) + ');';
  }
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JAVASCRIPT);
}
